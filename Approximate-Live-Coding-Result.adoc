class JRentService {
    private final Map<String, JRental> rentals;

    public JRentService(Map<String, JRental> rentals) {
        this.rentals = rentals;
    }

    public float amountOwed(String rentalId) {
        JRental rental = rentals.get(rentalId);
        if(rental == null)
            throw new IllegalArgumentException("Rental "+rentalId+" is missing");
        return effectiveRentedDays(rental);
    }

    private float effectiveRentedDays(JRental rental) {
        return switch (rental){
            case JChildrensRental(JDay daysRented) when daysRented.compareTo(2f) > 0
                    -> (daysRented.minus(2)).rawValue();
            case JChildrensRental _
                    -> 2;
            case JNewRental(JDay daysRented)
                    -> daysRented.times(10).rawValue();
        };
    }
}

record JDay(float rawValue) implements Comparable<Float> {
    @Override
    public int compareTo(@NonNull Float o) {
        return Float.compare(rawValue, o);
    }

    public JDay minus(float other){
        return new JDay(rawValue - other);
    }

    public JDay times(float other){
        return new JDay(rawValue - other);
    }
}

sealed interface JRental{ }

record JChildrensRental(JDay daysRented) implements JRental {}

record JNewRental(JDay daysRented) implements JRental {}

class KRentService(private val rentals: Map<String, KRental>) {

    fun amountOwed(rentalId: String): Float {
        val rental = rentals[rentalId]
            ?: throw IllegalArgumentException("Rental '$rentalId' does not exist")
        return effectiveRentedDays(rental).rawValue
    }

    private fun effectiveRentedDays(rental: KRental): KDay {
        return when(rental) {
            is KChildrensRental -> {
                val (daysRented) = rental
                if(daysRented > 2.days())
                    daysRented - 2.days()
                else
                    2.days()
            }
            is KNewRental -> rental.daysRented * 10f
        }
    }
}
